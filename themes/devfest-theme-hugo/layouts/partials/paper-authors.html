{{/* Renders a paper's author list, emphasising the presenting author(s). Presenters come from the data/presenters-YYYY.yml file generated by scripts/mark_presenters.py. Years without such a file simply render every author unemphasised. Expects a dict with: paper - the accepted-paper record currentYear - the conference year affiliations - whether to print each author's affiliation (optional) presenters - set to true to emphasise the presenting author. Off by default, so a caller has to opt in. Contributed talks and posters both do. They share one submission space, so one file can key both by paper id; mark_presenters.py refuses to write the file should those ids ever start to overlap. (optional) emphasis - "strong" (default) or "em". The schedule packs many talks together, where bold reads as too heavy, so its cards ask for italics. (optional) */}} {{ $paper := .paper }} {{ $affiliations := .affiliations }} {{ $italic := eq (.emphasis | default "strong") "em" }} {{ $presenters := slice }} {{ if eq .presenters true }} {{ $byYear := index site.Data (printf "presenters-%s" (string .currentYear)) | default dict }} {{ $presenters = index $byYear (printf "%v" (int $paper.pid)) | default slice }} {{ end }} {{ $authors := $paper.authors }}
{{- range $key, $author := $authors }} {{- $name := printf "%s %s" $author.first $author.last }} {{- if in $presenters $name -}} {{ if $italic }}{{ $name }}{{ else }}{{ $name }}{{ end }} {{- else }}{{ $name }}{{ end }} {{- if and $affiliations $author.affiliation }} ({{ $author.affiliation }}){{ end }} {{- if ne $key (sub (len $authors) 1) }}; {{ end }} {{- end }}
